-- Every narrative the classifier can actually score on the most recent run: both
-- indicators present, badge or no badge. This is the unbiased cohort — the earlier
-- alert_level_indicators.txt pull only returned narratives that already carried a
-- badge, which the old classifier had picked out of stale rows, so any distribution
-- measured on it is a distribution of the bug.
--
-- Finds the latest run itself (MAX(calculated_at::date)) rather than hardcoding a
-- date, so it doesn't depend on assumptions about what's deployed.
--
-- Both scales come back for each axis: indicator_value is the raw score, and
-- metadata->>'percentile' is its rank within the run's cohort. The redesign needs
-- both, because the four regions mean completely different things depending on
-- which one the axes are drawn in.
--
-- Expect a few thousand rows: acceleration is computed for ~22k narratives but
-- composite only for the prevalent set, and classification requires both.

WITH latest_run AS (
    SELECT MAX(calculated_at::date) AS run_day
    FROM narrative_analysis_indicators
    WHERE indicator_type = 'acceleration_rate'
),
ind AS (
    SELECT DISTINCT ON (narrative_id, indicator_type)
           narrative_id,
           indicator_type,
           indicator_value,
           (metadata->>'percentile')::float AS percentile
    FROM narrative_analysis_indicators
    WHERE calculated_at::date = (SELECT run_day FROM latest_run)
      AND indicator_type IN ('composite_virality', 'acceleration_rate')
    ORDER BY narrative_id, indicator_type, calculated_at DESC
)
SELECT
    i.narrative_id,
    n.alert_level,
    max(i.indicator_value) FILTER (WHERE i.indicator_type = 'composite_virality') AS composite,
    max(i.percentile)      FILTER (WHERE i.indicator_type = 'composite_virality') AS composite_pct,
    max(i.indicator_value) FILTER (WHERE i.indicator_type = 'acceleration_rate')  AS accel,
    max(i.percentile)      FILTER (WHERE i.indicator_type = 'acceleration_rate')  AS accel_pct
FROM ind i
JOIN narratives n ON n.id = i.narrative_id
GROUP BY i.narrative_id, n.alert_level
HAVING max(i.indicator_value) FILTER (WHERE i.indicator_type = 'composite_virality') IS NOT NULL
   AND max(i.indicator_value) FILTER (WHERE i.indicator_type = 'acceleration_rate')  IS NOT NULL
ORDER BY composite DESC;
