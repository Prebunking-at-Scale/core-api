-- Per-narrative measurement coverage on calc_date. Answers two things at once:
--
-- 1) D5's predicate: does the narrative have ANY video refreshed on calc_date?
--    Splits the 1732 narratives currently at accel = 0 into
--      - videos_refreshed > 0  -> genuinely FLAT, keep in the cohort, rank 0 honestly
--      - videos_refreshed = 0  -> UNMEASURED, exclude per D5
--    Every D6/D7/D8 number so far used `accel > 0` as a proxy, which wrongly dropped
--    the first group. This is what the real cohort looks like.
--
-- 2) The dilution question: acceleration's numerator only covers refreshed videos while
--    its denominator covers all of them, so it is biased down by (1 - coverage). If the
--    median mover sits at ~5% coverage, then its measured 0.46% growth is ~9% actual,
--    and the "movers are noise" reading is wrong.
--
-- coverage_by_views is the one that matters: one refreshed video holding 99% of the
-- narrative's views is 99% covered, not 1%. coverage_by_count is carried for contrast.
--
-- Set calc_date to the day the last run SCORED (not the day it ran). As of writing the
-- pipeline scores yesterday, so the 2026-07-16 00:05 run scored 2026-07-15.

WITH params AS (
    SELECT DATE '2026-07-15' AS calc_date,
           DATE '2026-07-14' AS prev_date
),
links AS (
    SELECT DISTINCT cn.narrative_id, vs.video_id
    FROM video_stats vs
    JOIN videos v         ON vs.video_id = v.id
    JOIN video_claims c   ON v.id = c.video_id
    JOIN claim_narratives cn ON c.id = cn.claim_id
),
-- baseline state entering calc_date: latest snapshot on or before prev_date
baseline AS (
    SELECT DISTINCT ON (l.narrative_id, l.video_id)
           l.narrative_id, l.video_id, vs.views AS baseline_views
    FROM links l
    JOIN video_stats vs ON vs.video_id = l.video_id
    WHERE vs.recorded_at::date <= (SELECT prev_date FROM params)
    ORDER BY l.narrative_id, l.video_id, vs.recorded_at DESC
),
-- videos that actually got a fresh snapshot ON calc_date
refreshed AS (
    SELECT DISTINCT l.narrative_id, l.video_id
    FROM links l
    JOIN video_stats vs ON vs.video_id = l.video_id
    WHERE vs.recorded_at::date = (SELECT calc_date FROM params)
),
-- how stale each video's baseline is, for the O5.4 max-gap question
gaps AS (
    SELECT DISTINCT ON (l.narrative_id, l.video_id)
           l.narrative_id, l.video_id,
           (SELECT calc_date FROM params) - vs.recorded_at::date AS baseline_age_days
    FROM links l
    JOIN video_stats vs ON vs.video_id = l.video_id
    WHERE vs.recorded_at::date <= (SELECT prev_date FROM params)
    ORDER BY l.narrative_id, l.video_id, vs.recorded_at DESC
)
SELECT
    b.narrative_id,

    count(*)                                                    AS videos,
    count(*) FILTER (WHERE r.video_id IS NOT NULL)              AS videos_refreshed,

    sum(b.baseline_views)                                       AS baseline_views,
    sum(b.baseline_views) FILTER (WHERE r.video_id IS NOT NULL) AS baseline_views_refreshed,

    -- the two coverage measures (O5.2)
    round(
        count(*) FILTER (WHERE r.video_id IS NOT NULL)::numeric
        / NULLIF(count(*), 0), 4
    )                                                           AS coverage_by_count,
    round(
        COALESCE(sum(b.baseline_views) FILTER (WHERE r.video_id IS NOT NULL), 0)::numeric
        / NULLIF(sum(b.baseline_views), 0), 4
    )                                                           AS coverage_by_views,

    -- baseline staleness among the refreshed videos (O5.4)
    max(g.baseline_age_days) FILTER (WHERE r.video_id IS NOT NULL) AS max_baseline_age_days,
    round(avg(g.baseline_age_days) FILTER (WHERE r.video_id IS NOT NULL), 2) AS avg_baseline_age_days
FROM baseline b
LEFT JOIN refreshed r ON r.narrative_id = b.narrative_id AND r.video_id = b.video_id
LEFT JOIN gaps g      ON g.narrative_id = b.narrative_id AND g.video_id = b.video_id
GROUP BY b.narrative_id;
