-- Current composite + acceleration (raw value AND percentile) for every narrative
-- carrying an alert badge. The percentile is what update_narrative_alert_levels
-- classifies on; indicator_value is what the UI displays. Showing both side by side
-- is the point: an early_surge row typically has accel ~0.0001 (displays as "0.00")
-- but accel_pct ~0.9996, because it sorts above a cohort that is ~99.8% exact zeros.
--
-- calculated_at is included per indicator: if composite and acceleration carry
-- different dates, the two halves of the classification came from different runs.

WITH latest AS (
    SELECT DISTINCT ON (narrative_id, indicator_type)
           narrative_id,
           indicator_type,
           indicator_value,
           (metadata->>'percentile')::float AS percentile,
           calculated_at
    FROM narrative_analysis_indicators
    WHERE indicator_type IN ('composite_virality', 'acceleration_rate')
    ORDER BY narrative_id, indicator_type, calculated_at DESC
)
SELECT
    n.id,
    left(n.title, 60) AS title,
    n.alert_level,

    max(l.indicator_value) FILTER (WHERE l.indicator_type = 'composite_virality') AS composite,
    max(l.percentile)      FILTER (WHERE l.indicator_type = 'composite_virality') AS composite_pct,

    max(l.indicator_value) FILTER (WHERE l.indicator_type = 'acceleration_rate')  AS accel,
    round(
        (max(l.indicator_value) FILTER (WHERE l.indicator_type = 'acceleration_rate'))::numeric, 2
    )                                                                             AS accel_rounded_as_ui,
    max(l.percentile)      FILTER (WHERE l.indicator_type = 'acceleration_rate')  AS accel_pct,

    max(l.calculated_at)   FILTER (WHERE l.indicator_type = 'composite_virality') AS composite_calculated_at,
    max(l.calculated_at)   FILTER (WHERE l.indicator_type = 'acceleration_rate')  AS accel_calculated_at
FROM narratives n
JOIN latest l ON l.narrative_id = n.id
WHERE n.alert_level IS NOT NULL
  AND n.alert_level <> 'none'
GROUP BY n.id, n.title, n.alert_level
ORDER BY
    array_position(
        ARRAY['viral', 'alert', 'early_surge', 'watch']::narrative_alert_level[],
        n.alert_level
    ),
    accel_pct DESC NULLS LAST;
